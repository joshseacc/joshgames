        function playSuccessSound() {

            osc.start();

            osc.stop(audioCtx.currentTime + 1.2);

        }

        function initCanvas() {

            const wrapper = document.getElementById('scratch-wrapper');

            const secretWord = document.getElementById('secret-word');

            canvas = document.getElementById('scratch-canvas');

            ctx = canvas.getContext('2d', { willReadFrequently: true });


            // 根據隱藏文字的實際大小設置畫布分辨率

            const rect = secretWord.getBoundingClientRect();

            canvas.width = rect.width;

            canvas.height = rect.height;


            // 繪製刮刮卡塗層 (銀灰色 + 簡單紋理)

            ctx.fillStyle = '#C0C0C0';

            // 添加一點暗色紋理讓它看起來像真的刮刮卡

            ctx.fillStyle = '#999999';

            for(let i=0; i<100; i++) {

                ctx.fillRect(Math.random() * canvas.width, Math.random() * canvas.height, 2, 2);

            }


            // 畫筆設置：destination-out 表示畫過的地方會變透明

            ctx.globalCompositeOperation = 'destination-out';

            ctx.lineJoin = 'round';

            ctx.lineCap = 'round';

            ctx.lineWidth = 25; // 橡皮擦粗細


            // 綁定事件

            isDrawing = false;

            

            // 滑鼠事件

            canvas.addEventListener('mousedown', startPosition);

            canvas.addEventListener('mouseup', endPosition);

            canvas.addEventListener('mousemove', draw);

            canvas.addEventListener('mouseleave', endPosition);

            

            // 觸控事件

            canvas.addEventListener('touchstart', startPosition, { passive: false });

            canvas.addEventListener('touchend', endPosition);

            canvas.addEventListener('touchmove', draw, { passive: false });

        }


        // 取得游標或手指在畫布上的相對座標

        function getPointerPos(e) {

            const rect = canvas.getBoundingClientRect();

            // 縮放比例 (防高解析度屏幕導致的偏移)

            const scaleX = canvas.width / rect.width;

            const scaleY = canvas.height / rect.height;

            

            let clientX, clientY;

            if (e.type.includes('touch')) {

                clientX = e.touches[0].clientX;

                clientY = e.touches[0].clientY;

            } else {

                clientX = e.clientX;

                clientY = e.clientY;

            }

            

            return {

                x: (clientX - rect.left) * scaleX,

                y: (clientY - rect.top) * scaleY

            };

        }


        function startPosition(e) {

            if (isScratched) return;

            e.preventDefault(); // 防止手機滾動

            isDrawing = true;

            draw(e); // 點擊時也刮除一點

        }


        function endPosition() {

            isDrawing = false;

            ctx.beginPath(); // 重置路徑

            

            // 每次抬起手時檢查刮除進度

            if (!isScratched) {

                checkScratchProgress();

            }

        }


        function draw(e) {

            if (!isDrawing || isScratched) return;

            e.preventDefault(); // 防止手機滾動


            const pos = getPointerPos(e);


            ctx.lineTo(pos.x, pos.y);

            ctx.stroke();

            ctx.beginPath();

            ctx.moveTo(pos.x, pos.y);


            // 每刮幾下檢查一次，避免頻繁計算導致卡頓

            scratchCheckCounter++;

            if (scratchCheckCounter > 15) {

                checkScratchProgress();

                scratchCheckCounter = 0;

            }

        }


        function checkScratchProgress() {

            const imageData = ctx.getImageData(0, 0, canvas.width, canvas.height);

            const pixels = imageData.data;

            let transparentPixels = 0;

            

            // 每 4 個值為一個像素 (r, g, b, a)，我們只檢查 Alpha 通道 (透明度)

            for (let i = 3; i < pixels.length; i += 4) {

                if (pixels[i] < 128) {

                    transparentPixels++;

                }

            }

            

            const totalPixels = pixels.length / 4;

            const percentage = (transparentPixels / totalPixels) * 100;


            // 如果刮開超過 50%，自動顯示全部

            if (percentage > 50) {

                revealAnswer();

            }

        }


        function revealAnswer() {

            // 顯示星星與文字

            successMsg.style.opacity = '1';

            

            // 重新觸發動畫

            const star = successMsg.firstElementChild;

            star.classList.remove('star-animate');

            void star.offsetWidth; // 強制重繪 (Magic trick for restarting animation)

            star.classList.add('star-animate');


            playSuccessSound();

        }

                document.getElementById('next-btn').classList.remove('hidden'); // 如果從完成畫面回來

                renderQuestion();

            }

        }


        function shuffleQuestions() {

            if (confirm('確定要打亂題目順序並從頭開始嗎？')) {

                // Fisher-Yates 洗牌演算法

                for (let i = questions.length - 1; i > 0; i--) {

                    const j = Math.floor(Math.random() * (i + 1));

                    [questions[i], questions[j]] = [questions[j], questions[i]];

                }

                currentIndex = 0;

                document.getElementById('next-btn').classList.remove('hidden');

                renderQuestion();

            }

        }
